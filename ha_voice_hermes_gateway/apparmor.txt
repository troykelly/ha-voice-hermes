#include <tunables/global>

profile ha_voice_hermes_gateway flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # The root S6/configuration supervisor has no network-facing parser. It must
  # initialize S6, safely snapshot mapped files, validate TLS, and drop the
  # gateway process to its dedicated account.
  file,
  network inet stream,
  network inet6 stream,
  signal (send) set=(kill,term,int,hup,cont),
  capability chown,
  capability dac_override,
  capability fowner,
  capability setgid,
  capability setuid,

  /init ix,
  /bin/** ix,
  /usr/bin/** ix,
  /usr/sbin/** ix,
  /run/{s6,s6-rc*,service}/** ix,
  /package/** ix,
  /command/** ix,
  /etc/services.d/** rwix,
  /etc/cont-init.d/** rwix,
  /etc/cont-finish.d/** rwix,
  /run/{,**} rwk,
  /dev/tty rw,
  /usr/lib/bashio/** ix,
  /tmp/{,**} rwk,

  /usr/local/bin/gateway-supervisor rix,
  /usr/local/bin/gateway-healthcheck rix,
  /usr/local/bin/prepare-runtime rix,
  /usr/local/bin/secure-snapshot rix,
  /usr/local/bin/validate-options.jq r,

  # Keep the network-facing runtime in a narrower child profile.
  /usr/local/bin/workerd cx -> workerd,

  profile workerd flags=(attach_disconnected,mediate_deleted) {
    #include <abstractions/base>
    #include <abstractions/nameservice>

    network inet stream,
    network inet dgram,
    network inet6 stream,
    network inet6 dgram,
    signal (receive) set=(kill,term,int,hup,cont) peer=*_ha_voice_hermes_gateway,

    /usr/local/bin/workerd rix,
    /opt/ha-voice-hermes/gateway/index.js r,
    /opt/ha-voice-hermes/gateway/index_bg.wasm r,
    /tmp/ha-voice-hermes/{,**} r,
    /data/workerd/{,**} rwk,
    /etc/ssl/certs/{,**} r,
    /dev/null rw,
    /dev/urandom r,
  }
}
